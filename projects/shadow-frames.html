FootstepComponent.cpp — Physical Material Detection with Line Trace 

FName UFootstepComponent::DetectSurfaceType() const
{
    FHitResult Hit;
    const FVector Start = GetOwner()->GetActorLocation();
    const FVector End = Start - FVector(0.f, 0.f, m_TraceDistance);

    FCollisionQueryParams QueryParams;
    QueryParams.AddIgnoredActor(GetOwner());
    QueryParams.bReturnPhysicalMaterial = true;

    if (GetWorld()->LineTraceSingleByChannel(Hit, Start, End, ECC_Visibility, QueryParams))
    {
        if (UPhysicalMaterial* PhysMat = Hit.PhysMaterial.Get())
        {
            // Map PhysicalSurface enum to Wwise Switch names
            switch (PhysMat->SurfaceType)
            {
                case SurfaceType1: return FName("Tiles");
                case SurfaceType2: return FName("Wood");
                case SurfaceType4: return FName("CreakyWood");
                default: return FName("Concrete");
            }
        }
    }
    return FName("Concrete");
}

PhotoboardWidget.cpp — Dynamic RTPC for Tension System 

void UPhotoboardWidget::UpdateAmbientRTPC()
{
    if (!PhotosOnBoardRTPC) return;

    // Find AkAmbientSound actor for progressive ambient system
    AActor* TargetActor = nullptr;
    TArray<AActor*> FoundActors;
    UGameplayStatics::GetAllActorsOfClass(
        GetWorld(),
        AAkAmbientSound::StaticClass(),
        FoundActors
    );

    for (AActor* Actor : FoundActors)
    {
        if (Actor && Actor->GetName().Contains(TEXT("AmbientProgressive")))
        {
            TargetActor = Actor;
            break;
        }
    }

    // RTPC value 0-5 drives ambient intensity as photos are placed
    UAkGameplayStatics::SetRTPCValue(
        PhotosOnBoardRTPC,
        m_ImportantPhotosAlreadyPlaced,
        0,
        TargetActor
    );
}
